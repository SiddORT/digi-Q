SELECT
  u.id AS receptionist_id,
  u.full_name,
  u.role,
  u.status,
  u.managing_admin_id,
  m.full_name AS managing_admin_name,
  a.clinic_id,
  c.data->>'name' AS clinic_name,
  c.admin_id AS clinic_owner_id,
  o.full_name AS clinic_owner_name,
  a.branch_id,
  b.status AS branch_status
FROM users u
LEFT JOIN users m ON m.id = u.managing_admin_id
LEFT JOIN assignments a ON a.user_id = u.id
LEFT JOIN clinics c ON c.id = a.clinic_id
LEFT JOIN users o ON o.id = c.admin_id
LEFT JOIN branches b ON b.id = a.branch_id
WHERE lower(trim(u.email)) = 'seema@gmail.com'
ORDER BY a.clinic_id, a.branch_id;
